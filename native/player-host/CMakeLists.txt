cmake_minimum_required(VERSION 3.24)
project(free_new_desk_player_host VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "player-host is supported only on Windows")
endif()
add_executable(player-host src/main.cpp src/mpv_loader.cpp src/named_pipe_server.cpp)
target_include_directories(player-host PRIVATE include)
target_compile_features(player-host PRIVATE cxx_std_20)
target_compile_definitions(player-host PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(player-host PRIVATE /W4 /permissive- /EHsc)
endif()
target_link_libraries(player-host PRIVATE kernel32 user32)
